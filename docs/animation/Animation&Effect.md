import React, { useState, useEffect } from "react"

const ValidationEffectsDemo = () => {

  const [isAnimating, setIsAnimating] = useState(true)

  // 샘플 데이터

    { id: 1, hospital: "서울대병원", type: "A", title: "일반검진", price: 150000, fee: null, status: "empty" },
    {
      id: 2,
      hospital: "연세세브란스",
      type: "B",
      title: "종합검진",
      price: 250000,
      fee: 15000,
      status: "duplicate",
      pair: [2, 5],
    },
    { id: 3, hospital: "삼성서울병원", type: "A", title: "정밀검진", price: 350000, fee: "", status: "empty" },
    { id: 4, hospital: "아산병원", type: "C", title: "특화검진", price: 180000, fee: 12000, status: "normal" },
    {
      id: 5,
      hospital: "고려대병원",
      type: "B",
      title: "종합검진",
      price: 250000,
      fee: 15000,
      status: "duplicate",
      pair: [2, 5],
    },
    { id: 6, hospital: "한양대병원", type: "A", title: "기본검진", price: null, fee: 8000, status: "empty" },
  ]

  const duplicatePairColors = [
    "border-l-red-500",
    "border-l-amber-500",
    "border-l-blue-500",
    "border-l-green-500",
    "border-l-purple-500",
    "border-l-pink-500",
  ]

  const getPairColor = (rowId) => {
    const row = sampleData.find((r) => r.id === rowId)
    if (row?.status === "duplicate" && row.pair) {
      const pairIndex = Math.min(...row.pair) - 1
      return duplicatePairColors[pairIndex % duplicatePairColors.length]
    }
    return ""
  }

  const CellComponent = ({ value, status, rowId, isPair = false }) => {
    const [isHovered, setIsHovered] = useState(false)
    const [shouldFlash, setShouldFlash] = useState(false)

    const handleNavigate = () => {
      setShouldFlash(true)
      setTimeout(() => setShouldFlash(false), 2000)
    }

    // 빈 값 효과 - 펄스/글로우 제거, 점선 테두리만 유지
    const emptyEffects = status === "empty" ? "bg-red-50 border-2 border-dashed border-red-300 text-red-600" : ""

    // 중복 효과
    const duplicateEffects =
      status === "duplicate"
        ? `bg-gradient-to-r from-amber-50 to-orange-50 ${getPairColor(rowId)} border-l-4 relative overflow-hidden`
        : ""

    // 플래시 효과
    const flashEffect = shouldFlash ? "animate-ping bg-blue-200" : ""

    return (
      <div
        className={`group relative mb-2 flex min-h-[60px] cursor-pointer items-center justify-between rounded-lg p-4 transition-all duration-300 hover:shadow-md ${emptyEffects} ${duplicateEffects} ${flashEffect} `}
        onMouseEnter={() => setIsHovered(true)}
        onMouseLeave={() => setIsHovered(false)}
        onClick={handleNavigate}>
        {/* 값 표시 */}
        <div className="flex flex-1 items-center gap-3">
          <span className={`font-medium ${status === "empty" ? "italic" : ""}`}>
            {value || (status === "empty" ? "비어있음" : "")}
          </span>

          {/* 빈 값 경고 아이콘 - 바운싱 제거, 정적 아이콘으로 */}
          {status === "empty" && (
            <div className="flex items-center gap-2">
              <div className="h-4 w-4 rounded-full bg-red-500"></div>
              <span className="rounded-full bg-red-200 px-2 py-1 text-xs font-medium text-red-800">필수</span>
            </div>
          )}
        </div>

        {/* 중복 표시 */}
        {status === "duplicate" && (
          <div className="flex items-center gap-2">
            <div className="flex items-center gap-1">
              <div className="h-4 w-4 animate-pulse rounded-full bg-amber-600"></div>
              <span className="rounded-full bg-amber-200 px-2 py-1 text-xs font-medium text-amber-800">2+</span>
            </div>

            {isHovered && (
              <div className="animate-fadeIn flex gap-1">
                <button className="transform rounded bg-amber-600 px-2 py-1 text-xs font-medium text-white transition-all hover:scale-105 hover:bg-amber-700">
                  #1
                </button>
                <button className="transform rounded bg-amber-600 px-2 py-1 text-xs font-medium text-white transition-all hover:scale-105 hover:bg-amber-700">
                  #2
                </button>
              </div>
            )}
          </div>
        )}

        {/* 호버 툴팁 */}
        {isHovered && (
          <div className="absolute bottom-full left-1/2 z-10 mb-2 -translate-x-1/2 transform">
            <div className="animate-fadeIn whitespace-nowrap rounded-lg bg-gray-900 px-3 py-2 text-xs text-white shadow-xl">
              {status === "empty" && "⚠️ 빈 값이 발견되었습니다"}
              {status === "duplicate" && "🔄 중복된 데이터입니다"}
              {status === "normal" && "✅ 정상 데이터입니다"}
              <div className="absolute left-1/2 top-full -translate-x-1/2 transform border-4 border-transparent border-t-gray-900"></div>
            </div>
          </div>
        )}
      </div>
    )
  }

  const NavigationPanel = () => {
    const [selectedGroup, setSelectedGroup] = useState(null)
    const duplicateGroups = [
      { id: 1, name: "종합검진 그룹", items: ["연세세브란스", "고려대병원"], color: "amber" },
      { id: 2, name: "정밀검진 그룹", items: ["삼성서울병원", "아산병원"], color: "blue" },
    ]

    return (
      <div className="animate-slideIn fixed right-4 top-4 z-40 w-full max-w-sm rounded-xl border border-gray-200 bg-white shadow-2xl">
        <div className="border-b border-gray-100 p-4">

                    <div className={`h-3 w-3 rounded-full bg-${group.color}-500`}></div>
                    <span className="font-medium text-amber-800">{group.name}</span>
                  </div>
                  <div className="flex items-center gap-2">
                    <span className="rounded-full bg-amber-200 px-2 py-1 text-xs font-medium text-amber-800">
                      {group.items.length}개
                    </span>
                    <span
                      className={`text-amber-600 transition-transform duration-200 ${selectedGroup === index ? "rotate-180" : ""}`}>
                      ▼
                    </span>
                  </div>
                </div>
              </div>

              {selectedGroup === index && (
                <div className="animate-fadeIn space-y-2 border-t border-amber-200 bg-white px-3 pb-3">
                  <div className="pt-2 text-xs font-medium text-amber-700">중복된 행들:</div>
                  {group.items.map((item, idx) => (
                    <button
                      key={idx}
                      className="w-full transform rounded border border-amber-200 bg-amber-50 p-2 text-left text-sm transition-all hover:scale-[1.02] hover:bg-amber-100 hover:shadow-sm">
                      <div className="flex items-center justify-between">
                        <span>{item}</span>
                        <span className="text-amber-600">→</span>
                      </div>
                    </button>
                  ))}
                </div>
              )}
            </div>
          ))}
        </div>
      </div>
    )
  }

  const StatusIndicator = () => {
    const [currentStatus, setCurrentStatus] = useState("processing")

    useEffect(() => {
      const interval = setInterval(() => {
        setCurrentStatus((prev) => {
          switch (prev) {
            case "processing":
              return "warning"
            case "warning":
              return "success"
            case "success":
              return "error"
            case "error":
              return "processing"
            default:
              return "processing"
          }
        })
      }, 2000)

      return () => clearInterval(interval)
    }, [])

    const statusConfig = {
      processing: { color: "bg-blue-500", text: "검증 중...", animation: "animate-spin" },
      success: { color: "bg-green-500", text: "검증 완료 ✓", animation: "animate-pulse" },
      warning: { color: "bg-yellow-500", text: "경고 3개 발견", animation: "animate-bounce" },
      error: { color: "bg-red-500", text: "오류 발견!", animation: "animate-ping" },
    }

    return (
      <div className="animate-slideUp fixed bottom-6 right-6 z-30 rounded-full border border-gray-200 bg-white p-4 shadow-lg">
        <div className="flex items-center gap-3">
          <div className={`h-3 w-3 rounded-full ${config.color} ${config.animation}`}></div>
          <span className="text-sm font-medium text-gray-700">{config.text}</span>
          <div className={`h-4 w-4 ${config.color} rounded-full opacity-50 ${config.animation}`}></div>
        </div>
      </div>
    )
  }

  return (

            <button
              key={filter}
              onClick={() => setActiveEffect(filter)}
              className={`transform rounded-lg px-4 py-2 font-medium transition-all hover:scale-105 ${
                activeEffect === filter
                  ? "bg-blue-600 text-white shadow-lg"
                  : "border border-gray-200 bg-white text-gray-700 hover:bg-gray-50"
              }`}>
              {filter === "all" && "전체"}
              {filter === "empty" && "빈 값"}
              {filter === "duplicate" && "중복"}
              {filter === "normal" && "정상"}
            </button>
          ))}
        </div>

        {/* 데이터 테이블 */}

            <div className="grid grid-cols-5 gap-4 text-sm font-semibold uppercase tracking-wider text-gray-900">

              <div>검진구분</div>
              <div>검진유형</div>

              <div>수수료</div>
            </div>
          </div>

                  key={row.id}
                  className={`grid grid-cols-5 gap-4 ${getPairColor(row.id)} ${row.status === "duplicate" ? "border-l-4" : ""}`}>

                  <CellComponent value={row.type} status={row.status} rowId={row.id} />
                  <CellComponent value={row.title} status={row.status} rowId={row.id} />

                  <CellComponent value={row.fee?.toLocaleString()} status={row.status} rowId={row.id} />

              ))}

### **Week 1-2: 기초 다지기**

3. **Slide Animation** - 슬라이드 전환 효과
4. **Modal Animation** - 모달 등장/사라짐

### **Week 5-6: 고급 효과**

1. **Parallax Animation** - 스크롤 기반 시차
2. **Morphing Animation** - 형태 변환
3. **Particle Animation** - 입자 효과
4. **Spring Animation** - 물리 기반 탄성
